-- Arquitetura do Banco de Dados: jhe_portfolio_db_2.0
-- Este script DDL é compatível com phpMyAdmin (MySQL).

CREATE DATABASE IF NOT EXISTS jhe_portfolio_db_2 CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE jhe_portfolio_db_2;

-- 1. Tabela: usuarios
-- Armazena os usuários do sistema, suportando roles para controle de acesso.
CREATE TABLE IF NOT EXISTS usuarios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(150) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL,
    role ENUM('admin_master', 'user') NOT NULL DEFAULT 'user',
    ativo BOOLEAN NOT NULL DEFAULT TRUE,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 2. Tabela: clientes
-- Clientes que possuem projetos registrados.
CREATE TABLE IF NOT EXISTS clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    logo_url VARCHAR(255),
    setor VARCHAR(100),
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 3. Tabela: projetos
-- Portfólio de projetos realizados.
CREATE TABLE IF NOT EXISTS projetos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    titulo VARCHAR(150) NOT NULL,
    cliente_id INT NOT NULL,
    setor VARCHAR(100),
    descricao TEXT,
    imagem_url VARCHAR(255),
    link_oficial VARCHAR(255),
    status VARCHAR(50) DEFAULT 'concluido',
    created_by INT,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    atualizado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id) ON DELETE CASCADE,
    FOREIGN KEY (created_by) REFERENCES usuarios(id) ON DELETE SET NULL
);

-- 4. Tabela: servicos
-- Serviços oferecidos.
CREATE TABLE IF NOT EXISTS servicos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT,
    icone VARCHAR(100),
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 5. Tabela: tecnologias
-- Tecnologias utilizadas.
CREATE TABLE IF NOT EXISTS tecnologias (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    icone_url VARCHAR(255),
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 6. Tabela: config_home
-- Configurações gerais da página inicial.
CREATE TABLE IF NOT EXISTS config_home (
    id INT AUTO_INCREMENT PRIMARY KEY,
    chave VARCHAR(100) NOT NULL UNIQUE,
    valor TEXT NOT NULL,
    atualizado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- 7. Tabela: telemetria_sessoes
-- Registro de acessos e sessões de usuários/visitantes.
CREATE TABLE IF NOT EXISTS telemetria_sessoes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ip_hash VARCHAR(255) NOT NULL,
    user_agent TEXT,
    data_acesso TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- 8. Tabela: telemetria_cliques
-- Registro de cliques em Call-to-Actions ou visualização de projetos.
CREATE TABLE IF NOT EXISTS telemetria_cliques (
    id INT AUTO_INCREMENT PRIMARY KEY,
    identificador_cta VARCHAR(100) NOT NULL,
    tipo VARCHAR(50),
    projeto_id INT,
    criado_em TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (projeto_id) REFERENCES projetos(id) ON DELETE SET NULL
);

-- 9. Tabela: auditoria_logs
-- Registro de ações críticas dentro do sistema administrativo.
CREATE TABLE IF NOT EXISTS auditoria_logs (
    id INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id INT,
    acao VARCHAR(100) NOT NULL,
    tabela_afetada VARCHAR(100),
    registro_id INT,
    ip_originario VARCHAR(100),
    status VARCHAR(50),
    detalhes TEXT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id) REFERENCES usuarios(id) ON DELETE SET NULL
);
